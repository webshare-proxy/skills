# Node.js Scraper Templates

## Dependencies

Standard mode:
```bash
npm init -y
npm install playwright dotenv
npx playwright install chromium
```

For stealth mode, use `patchright` instead of `playwright`:
```bash
npm init -y
npm install patchright dotenv
npx patchright install chromium
```

For SQLite storage, also install:
```bash
npm install better-sqlite3
```

## Config Module

```javascript
// config/config.js
require("dotenv").config();

const config = {
  proxy: {
    server: process.env.PROXY_SERVER || "",
    username: process.env.PROXY_USERNAME || "",
    password: process.env.PROXY_PASSWORD || "",
  },
  headless: (process.env.HEADLESS || "true") === "true",
  requestDelay: parseFloat(process.env.REQUEST_DELAY || "1.0") * 1000,
  maxPages: parseInt(process.env.MAX_PAGES || "10", 10),
  viewport: { width: 1920, height: 1080 },
};

function getProxyConfig() {
  if (!config.proxy.server) return undefined;
  return config.proxy;
}

module.exports = { config, getProxyConfig };
```

## Browser Utility

```javascript
// utils/browser.js
let playwright;
try {
  playwright = require("patchright");
} catch {
  playwright = require("playwright");
}

const { config, getProxyConfig } = require("../config/config");

async function createBrowser() {
  const proxy = getProxyConfig();
  const browser = await playwright.chromium.launch({
    headless: config.headless,
    proxy,
  });
  const context = await browser.newContext({
    userAgent: config.userAgent,
    viewport: config.viewport,
    locale: "en-US",
  });
  return { browser, context };
}

async function createPage(context) {
  const page = await context.newPage();
  page.setDefaultTimeout(30000);
  return page;
}

module.exports = { createBrowser, createPage, playwright };
```

## Storage: CSV

```javascript
// utils/storage.js - CSV
const fs = require("fs");
const path = require("path");

function saveToCsv(data, outputDir = "output/csv") {
  fs.mkdirSync(outputDir, { recursive: true });
  if (!data.length) return;

  const timestamp = new Date().toISOString().replace(/[:.]/g, "").slice(0, 15);
  const filepath = path.join(outputDir, `scrape_${timestamp}.csv`);
  const headers = Object.keys(data[0]);
  const csvRows = [headers.join(",")];

  for (const item of data) {
    const row = headers.map((h) => {
      const val = String(item[h] ?? "");
      // Escape CSV values containing commas, quotes, or newlines
      if (val.includes(",") || val.includes('"') || val.includes("\n")) {
        return `"${val.replace(/"/g, '""')}"`;
      }
      return val;
    });
    csvRows.push(row.join(","));
  }

  fs.writeFileSync(filepath, csvRows.join("\n"), "utf-8");
  console.log(`Saved ${data.length} items to ${filepath}`);
}

module.exports = { saveToCsv };
```

## Storage: SQLite (using better-sqlite3)

```javascript
// utils/storage.js - SQLite
const Database = require("better-sqlite3");
const fs = require("fs");
const path = require("path");

function saveToDb(data, dbPath = "output/db/scraper.db", tableName = "scraped_data") {
  fs.mkdirSync(path.dirname(dbPath), { recursive: true });
  if (!data.length) return;

  const db = new Database(dbPath);
  const columns = Object.keys(data[0]);

  const colDefs = columns.map((col) => `"${col}" TEXT`).join(", ");
  db.exec(
    `CREATE TABLE IF NOT EXISTS "${tableName}" ` +
      `(id INTEGER PRIMARY KEY AUTOINCREMENT, ` +
      `scraped_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, ${colDefs})`
  );

  const placeholders = columns.map(() => "?").join(", ");
  const colNames = columns.map((col) => `"${col}"`).join(", ");
  const stmt = db.prepare(
    `INSERT INTO "${tableName}" (${colNames}) VALUES (${placeholders})`
  );

  const insertMany = db.transaction((items) => {
    for (const item of items) {
      const values = columns.map((col) => String(item[col] ?? ""));
      stmt.run(...values);
    }
  });

  insertMany(data);
  db.close();
  console.log(`Saved ${data.length} items to ${dbPath} (table: ${tableName})`);
}

module.exports = { saveToDb };
```

## package.json Setup

Initialize and install deps via commands rather than hardcoding versions:

```bash
npm init -y
npm install playwright dotenv
# or for stealth: npm install patchright dotenv
# for SQLite: npm install better-sqlite3
```

Then add these scripts to the generated `package.json`:

```json
{
  "scripts": {
    "scrape": "node scrapers/example.js",
    "install-browser": "npx playwright install chromium"
  }
}
```

For stealth mode, change the install-browser script to `npx patchright install chromium`.
